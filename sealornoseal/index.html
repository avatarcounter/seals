// --- CONFIGURATION --- const PUBLIC_KEY = "69f60e2d8f40bb1068b944a4"; const TOTAL_SEALS = 5; const TOTAL_NOSEALS = 5; const EXTENSION = "jpg"; // --------------------- let isCorrectSeal = false; let score = 0; let gameActive = true; // 1. FETCH GLOBAL SCORES (Using Public Key) async function updateLeaderboardUI() { const list = document.getElementById('leaderboard-list'); try { const response = await fetch(`https://dreamlo.com{PUBLIC_KEY}/json`); const data = await response.json(); if (!data.dreamlo.leaderboard || !data.dreamlo.leaderboard.entry) { list.innerText = "Be the first to score!"; return; } let entries = data.dreamlo.leaderboard.entry; if (!Array.isArray(entries)) entries = [entries]; // Handle single entry case list.innerHTML = entries.slice(0, 5).map(s => `
${s.name} ${s.score}
`).join(''); } catch (e) { list.innerText = "Error loading leaderboard."; } } // 2. SAVE SCORE (Using Netlify Function to hide Private Key) function saveScore(name, finalScore) { const cleanName = name.replace(/\s+/g, '').substring(0, 12); fetch('/.netlify/functions/save-score', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ name: cleanName, score: finalScore }) }) .then(() => { // Refresh the list after a short delay to allow DB update setTimeout(updateLeaderboardUI, 1200); }) .catch(err => console.error("Save error:", err)); } // 3. GAME LOGIC function nextRound() { gameActive = true; const feedback = document.getElementById('feedback'); const imgElement = document.getElementById('display-img'); feedback.innerText = ""; isCorrectSeal = Math.random() < 0.5; const folder = isCorrectSeal ? "seal" : "noseal"; const max = isCorrectSeal ? TOTAL_SEALS : TOTAL_NOSEALS; const randomNum = Math.floor(Math.random() * max) + 1; // Formats number to 001, 002, etc. const paddedNum = randomNum.toString().padStart(3, '0'); imgElement.src = `${folder}/${paddedNum}.${EXTENSION}`; } function check(userChoice) { if (!gameActive) return; gameActive = false; const feedback = document.getElementById('feedback'); const scoreDisplay = document.getElementById('correct-count'); if (userChoice === isCorrectSeal) { score++; scoreDisplay.innerText = score; feedback.innerText = "Correct! 🌊"; feedback.style.color = "#27ae60"; setTimeout(nextRound, 1000); } else { feedback.innerText = `Game Over! That was a ${isCorrectSeal ? "Seal" : "Sea Lion"}.`; feedback.style.color = "#c0392b"; setTimeout(() => { if (score > 0) { const name = prompt(`Streak: ${score}! Enter name for the World Leaderboard:`); if (name && name.trim()) saveScore(name, score); } score = 0; scoreDisplay.innerText = 0; nextRound(); }, 1000); } } // Start game and load scores on page load updateLeaderboardUI(); nextRound();
